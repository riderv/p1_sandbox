cmake_minimum_required(VERSION 3.16)
project(p1_sandbox C)

# Устанавливаем современный стандарт Си
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)

# 1. Просим CMake автоматически найти Raylib в системных путях MSYS2 UCRT64
find_package(raylib REQUIRED)

# 2. Объявляем исполняемый файл p1 и включаем его файлы в дерево проекта
add_executable(p1 main.c pch.h JetBrainsMono-SemiBoldItalic.ttf)

# 3. Подключаем ваш предкомпилированный заголовок pch.h
target_precompile_headers(p1 PRIVATE pch.h)

# 4. Линкуем статическую библиотеку Raylib к проекту p1
# CMake автоматически добавит все необходимые системные зависимости Windows (opengl32, gdi32, winmm)
target_link_libraries(p1 PRIVATE raylib)

# ДОБАВЬТЕ ЭТУ СТРОКУ: Помогает clangd гарантированно найти raylib.h
target_include_directories(p1 PRIVATE C:/msys64/ucrt64/include)
